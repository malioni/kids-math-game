# Audio Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add feedback sounds and a quiet background loop to every world and the menu, with two saved
toggles in the level picker, using generated placeholder sounds.

**Architecture:** An `AudioManager` autoload plays sounds by name (`"win"`, `"forest/place"`) from
`assets/audio/`, on two buses (Music, Effects) that the picker toggles mute. `WorldBase`, `MainMenu` and
`LevelHud` call it in response to signals; mechanics only gain `piece_placed` / `piece_removed` signals and
never touch audio. A headless tool generates the placeholder `.wav` files.

**Tech Stack:** Godot 4.6.2, GDScript, GUT 9.6.0, gdtoolkit 4.5.0.

**Spec:** `docs/superpowers/specs/2026-11-09-audio-design.md`

## Global Constraints

- Sound names map to `res://assets/audio/<name>.wav`.
- Shared sounds: `win`, `lose`, `complete`, `tap`. Each world: `<world>/ambience`, `<world>/place`, `<world>/remove`. Menu: `menu/ambience`.
- Placeholder files: 22,050 Hz, mono, 16-bit, generated by `tools/make_audio_placeholders.gd`.
- Effects use a pool of 6 players. Background loops crossfade over 0.5 s, and asking for the current loop does nothing.
- Buses: `Master` with children `Music` and `Effects`; the toggles mute a bus.
- Settings saved in the `meta` section as `music_enabled` / `effects_enabled`, default on; an invalid value loads as on.
- A missing or unknown sound warns once and stays silent. Nothing awaits a sound.
- Mechanics never reference audio.
- New buttons are at least 100 px tall (picker buttons use `Vector2(120, 100)`).
- Code style from `CLAUDE.md`: type hints everywhere, `##` doc comments on public items, signals first, `_private` names.

## Review Focus

- **Fast tapping:** a child places bags quickly; every tap should sound and none cut off the previous one. Test in Task 3.
- **Quick back-and-forth between menu and world:** switching loops again mid-crossfade should end with only the last loop audible. Test in Task 3.
- **Sound turned off last time:** the game should start silent, not play until the setting is read. Test in Task 3.
- **Long play in one world:** the background loop should keep going past its 8 s file. Test in Task 3.
- **Rejected moves:** a plank too long for the gap, a full row, a drop off the table and a tap on an unplaced piece should make no place or remove sound. Tests in Task 4.

- [ ] **Step 1: Write the sound tool** — create `tools/make_audio_placeholders.gd`:

```gdscript
extends SceneTree

## Generates the placeholder sounds into assets/audio/: shared feedback sounds, each world's place,
## remove and background loop, and the menu's loop. Every sound is built in code, so nothing is
## downloaded or licensed; real sounds can replace them later under the same names. Run from the
## project root:
##   Godot --path . --headless -s tools/make_audio_placeholders.gd
## then `Godot --path . --headless --import` to create the .import files.

const OUT := "res://assets/audio/"
const RATE := 22050
## Length of each background loop, in seconds.
const LOOP_SECONDS := 8.0
## Extra audio generated past the loop's end and blended into its start, so it repeats without a gap.
const BLEND_SECONDS := 1.0
## Pitch, in Hz, of each world's place sound; remove is higher and lighter.
const PLACE_HZ := {"forest": 180.0, "sky": 520.0, "market": 140.0, "picnic": 330.0}

var _rng := RandomNumberGenerator.new()


func _init() -> void:
	_rng.seed = 8
	_save("win", _notes([523.25, 659.25, 783.99], 0.11, 0.45))
	_save("complete", _notes([523.25, 659.25, 783.99, 1046.5, 1318.5], 0.13, 0.7))
	_save("lose", _thud(150.0, 85.0, 0.35, 0.7))
	_save("tap", _blip(1100.0, 0.06, 0.35))
	for world: String in PLACE_HZ:
		var hz: float = PLACE_HZ[world]
		_save(world + "/place", _thud(hz, hz * 0.7, 0.14, 0.75))
		_save(world + "/remove", _blip(hz * 2.5, 0.08, 0.4))
	_save("sky/remove", _pop())
	_save("forest/ambience", _seamless(_water()))
	_save("sky/ambience", _seamless(_wind()))
	_save("market/ambience", _seamless(_murmur()))
	_save("picnic/ambience", _seamless(_birds()))
	_save("menu/ambience", _seamless(_pad()))
	quit()


func _save(sound: String, samples: PackedFloat32Array) -> void:
	var data := PackedByteArray()
	data.resize(samples.size() * 2)
	for i in samples.size():
		data.encode_s16(i * 2, int(clampf(samples[i], -1.0, 1.0) * 32767.0))
	var wav := AudioStreamWAV.new()
	wav.format = AudioStreamWAV.FORMAT_16_BITS
	wav.mix_rate = RATE
	wav.stereo = false
	wav.data = data
	var path: String = ProjectSettings.globalize_path(OUT + sound + ".wav")
	DirAccess.make_dir_recursive_absolute(path.get_base_dir())
	if wav.save_to_wav(path) != OK:
		push_error("Could not save %s" % path)


func _empty(seconds: float) -> PackedFloat32Array:
	var samples := PackedFloat32Array()
	samples.resize(int(seconds * RATE))
	return samples


# Adds a soft bell-like note starting at start seconds and fading out over length seconds.
func _add_note(
	samples: PackedFloat32Array, hz: float, start: float, length: float, gain: float
) -> void:
	var first: int = int(start * RATE)
	for i in int(length * RATE):
		if first + i >= samples.size():
			return
		var t: float = float(i) / RATE
		var envelope: float = minf(t / 0.005, 1.0) * exp(-4.0 * t / length)
		var wave: float = sin(TAU * hz * t) + 0.3 * sin(TAU * hz * 2.0 * t)
		samples[first + i] += wave * envelope * gain


# Notes played one after another, gap seconds apart, each ringing for tail seconds.
func _notes(hz: Array, gap: float, tail: float) -> PackedFloat32Array:
	var samples := _empty(gap * hz.size() + tail)
	for k in hz.size():
		_add_note(samples, hz[k], k * gap, tail, 0.35)
	return samples


# A short drop in pitch with a little noise: a soft knock or thud.
func _thud(from_hz: float, to_hz: float, length: float, gain: float) -> PackedFloat32Array:
	var samples := _empty(length)
	var phase: float = 0.0
	for i in samples.size():
		var t: float = float(i) / RATE
		var k: float = t / length
		phase += TAU * lerpf(from_hz, to_hz, k) / RATE
		var envelope: float = minf(t / 0.003, 1.0) * pow(1.0 - k, 3.0)
		samples[i] = (sin(phase) + 0.15 * _rng.randf_range(-1.0, 1.0)) * envelope * gain
	return samples


# A short, quickly fading tone: a click.
func _blip(hz: float, length: float, gain: float) -> PackedFloat32Array:
	var samples := _empty(length)
	for i in samples.size():
		var t: float = float(i) / RATE
		var envelope: float = minf(t / 0.002, 1.0) * exp(-6.0 * t / length)
		samples[i] = sin(TAU * hz * t) * envelope * gain
	return samples


# A burst of noise over a falling tone: a balloon popping.
func _pop() -> PackedFloat32Array:
	var samples := _empty(0.09)
	var phase: float = 0.0
	for i in samples.size():
		var t: float = float(i) / RATE
		phase += TAU * lerpf(900.0, 300.0, t / 0.09) / RATE
		var envelope: float = exp(-t / 0.018)
		samples[i] = (0.7 * _rng.randf_range(-1.0, 1.0) + 0.4 * sin(phase)) * envelope * 0.8
	return samples


# Blends the audio past LOOP_SECONDS into the start, so the end runs straight into the beginning.
func _seamless(raw: PackedFloat32Array) -> PackedFloat32Array:
	var length: int = int(LOOP_SECONDS * RATE)
	var blend: int = int(BLEND_SECONDS * RATE)
	var samples: PackedFloat32Array = raw.slice(0, length)
	for i in blend:
		var k: float = float(i) / blend
		samples[i] = raw[i] * k + raw[length + i] * (1.0 - k)
	return samples


func _loop_buffer() -> PackedFloat32Array:
	return _empty(LOOP_SECONDS + BLEND_SECONDS)


# Scales samples so their loudest point is peak.
func _normalize(samples: PackedFloat32Array, peak: float) -> PackedFloat32Array:
	var loudest: float = 0.0001
	for v in samples:
		loudest = maxf(loudest, absf(v))
	for i in samples.size():
		samples[i] *= peak / loudest
	return samples


# Low-passed noise that rises and falls a little: a river.
func _water() -> PackedFloat32Array:
	var samples := _loop_buffer()
	var low: float = 0.0
	var bubbles: float = 0.0
	for i in samples.size():
		var t: float = float(i) / RATE
		var noise: float = _rng.randf_range(-1.0, 1.0)
		low += 0.05 * (noise - low)
		bubbles += 0.3 * (noise - bubbles)
		var swell: float = 0.75 + 0.25 * sin(TAU * t * 0.3)
		samples[i] = (low * 3.0 + bubbles * 0.25) * swell
	return _normalize(samples, 0.5)


# Heavily low-passed noise with slow swells: wind.
func _wind() -> PackedFloat32Array:
	var samples := _loop_buffer()
	var low: float = 0.0
	for i in samples.size():
		var t: float = float(i) / RATE
		low += 0.015 * (_rng.randf_range(-1.0, 1.0) - low)
		samples[i] = low * (0.55 + 0.45 * sin(TAU * t / 4.5))
	return _normalize(samples, 0.5)


# Band-limited noise in short random bursts, like far-off voices: a market.
func _murmur() -> PackedFloat32Array:
	var samples := _loop_buffer()
	var low: float = 0.0
	var lower: float = 0.0
	var level: float = 0.0
	var target: float = 0.0
	var next_change: int = 0
	for i in samples.size():
		if i >= next_change:
			target = _rng.randf_range(0.2, 1.0)
			next_change = i + int(_rng.randf_range(0.15, 0.4) * RATE)
		level += 0.002 * (target - level)
		var noise: float = _rng.randf_range(-1.0, 1.0)
		low += 0.15 * (noise - low)
		lower += 0.02 * (noise - lower)
		samples[i] = (low - lower) * level
	return _normalize(samples, 0.4)


# A soft breeze with short rising chirps at random times: birds at a picnic.
func _birds() -> PackedFloat32Array:
	var samples := _loop_buffer()
	var low: float = 0.0
	for i in samples.size():
		low += 0.01 * (_rng.randf_range(-1.0, 1.0) - low)
		samples[i] = low * 0.6
	var start: float = 0.3
	while start < LOOP_SECONDS + BLEND_SECONDS - 0.5:
		for note in _rng.randi_range(2, 3):
			_add_chirp(samples, start + note * 0.14)
		start += _rng.randf_range(0.8, 2.0)
	return _normalize(samples, 0.45)


func _add_chirp(samples: PackedFloat32Array, start: float) -> void:
	var first: int = int(start * RATE)
	var length: int = int(0.1 * RATE)
	var phase: float = 0.0
	var from_hz: float = _rng.randf_range(2300.0, 2800.0)
	for i in length:
		if first + i >= samples.size():
			return
		var k: float = float(i) / length
		phase += TAU * lerpf(from_hz, from_hz * 1.35, k) / RATE
		samples[first + i] += sin(phase) * sin(PI * k) * 0.25


# Three quiet tones with a slow wobble: a calm menu loop.
func _pad() -> PackedFloat32Array:
	var samples := _loop_buffer()
	for i in samples.size():
		var t: float = float(i) / RATE
		var wobble: float = 0.8 + 0.2 * sin(TAU * t * 0.25)
		samples[i] = (
			(sin(TAU * 130.81 * t) + 0.7 * sin(TAU * 196.0 * t) + 0.5 * sin(TAU * 261.63 * t))
			* wobble
		)
	return _normalize(samples, 0.3)
```

func _arc(image: Image, center: Vector2, inner: float, outer: float, color: Color) -> void:
	for y in SIZE:
		for x in SIZE:
			var d: Vector2 = Vector2(x + 0.5, y + 0.5) - center
			if d.x > 6.0 and absf(d.angle()) < 0.9 and d.length() >= inner and d.length() <= outer:
				image.set_pixel(x, y, color)

	return AudioServer.is_bus_mute(AudioServer.get_bus_index(bus))


func test_audio_manager_every_expected_sound_loads() -> void:

```gdscript
extends Node

## Plays sound effects and background loops by name. A name maps to a file in assets/audio/:
## "win" is win.wav, "forest/place" is forest/place.wav. Effects play on the Effects bus and loops on
## the Music bus, and a grown-up can turn each off in the level picker; the choice is saved. A missing
## sound warns once and stays silent, so a sound problem never stops the game; a test checks every
## expected sound exists.
##
## Mechanics never call this. WorldBase, the main menu and the level HUD do, in response to signals.
## If another system (adaptive difficulty, a parent view) comes to need the same moments, add a
## game-events autoload that WorldBase reports to and have this listen to it, rather than adding more
## direct calls.

## Emitted when an effect starts.
signal sfx_played(sound: String)
## Emitted when a different background loop starts.
signal music_changed(sound: String)

## Sounds every world shares.
const SHARED_SOUNDS: Array[String] = ["win", "lose", "complete", "tap"]
## Sounds each world needs under its id, e.g. "forest/place".
const WORLD_SOUNDS: Array[String] = ["ambience", "place", "remove"]
## The main menu's background loop.
const MENU_MUSIC := "menu/ambience"

const _AUDIO_DIR := "res://assets/audio/"
const _MUSIC_BUS := &"Music"
const _EFFECTS_BUS := &"Effects"
const _EFFECT_PLAYERS := 6
const _CROSSFADE := 0.5
const _MUSIC_DB := -10.0
const _SILENT_DB := -60.0

var _streams := {}
var _warned := {}
var _effect_players: Array[AudioStreamPlayer] = []
var _next_effect: int = 0
# The loop playing now is _music_players[0]; the other fades out or waits.
var _music_players: Array[AudioStreamPlayer] = []
var _music: String = ""
var _fade: Tween


func _ready() -> void:
	for i in _EFFECT_PLAYERS:
		_effect_players.append(_add_player(_EFFECTS_BUS))
	for i in 2:
		_music_players.append(_add_player(_MUSIC_BUS))
	apply_saved_settings()


## Plays a sound effect by name. Players are taken in turn, so quick sounds overlap rather than cut
## each other off. An unknown name stays silent.
func play_sfx(sound: String) -> void:
	var stream: AudioStream = _stream(sound)
	if stream == null:
		return
	var player: AudioStreamPlayer = _effect_players[_next_effect]
	_next_effect = (_next_effect + 1) % _effect_players.size()
	player.stream = stream
	player.play()
	sfx_played.emit(sound)


## Crossfades to the named background loop. Asking for the loop already playing does nothing, and an
## unknown name leaves the current loop playing.
func play_music(sound: String) -> void:
	if sound == _music:
		return
	var stream: AudioStream = _stream(sound)
	if stream == null:
		return
	_set_looping(stream)
	_music = sound
	var outgoing: AudioStreamPlayer = _music_players[0]
	var incoming: AudioStreamPlayer = _music_players[1]
	_music_players = [incoming, outgoing]
	if _fade:
		_fade.kill()
	incoming.stream = stream
	incoming.volume_db = _SILENT_DB
	incoming.play()
	_fade = create_tween().set_parallel()
	_fade.tween_property(incoming, "volume_db", _MUSIC_DB, _CROSSFADE)
	_fade.tween_property(outgoing, "volume_db", _SILENT_DB, _CROSSFADE)
	_fade.chain().tween_callback(outgoing.stop)
	music_changed.emit(sound)


## Returns the name of the background loop playing, or "" if none.
func current_music() -> String:
	return _music


## Returns every sound name the game expects to find: the shared sounds, each registered world's
## WORLD_SOUNDS, and the menu's loop.
func expected_sounds() -> Array[String]:
	var sounds: Array[String] = SHARED_SOUNDS.duplicate()
	for world in GameRegistry.get_worlds():
		for sound in WORLD_SOUNDS:
			sounds.append("%s/%s" % [world["id"], sound])
	sounds.append(MENU_MUSIC)
	return sounds


## Turns background sound on or off, and saves the choice.
func set_music_enabled(enabled: bool) -> void:
	SaveManager.save_music_enabled(enabled)
	_mute_bus(_MUSIC_BUS, not enabled)


## Returns true while background sound is on.
func is_music_enabled() -> bool:
	return SaveManager.load_music_enabled()


## Turns sound effects on or off, and saves the choice.
func set_effects_enabled(enabled: bool) -> void:
	SaveManager.save_effects_enabled(enabled)
	_mute_bus(_EFFECTS_BUS, not enabled)


## Returns true while sound effects are on.
func is_effects_enabled() -> bool:
	return SaveManager.load_effects_enabled()


## Mutes or unmutes each bus to match the saved settings. Runs at startup, so sound turned off last
## time is off before anything plays.
func apply_saved_settings() -> void:
	_mute_bus(_MUSIC_BUS, not SaveManager.load_music_enabled())
	_mute_bus(_EFFECTS_BUS, not SaveManager.load_effects_enabled())


func _add_player(bus: StringName) -> AudioStreamPlayer:
	var player := AudioStreamPlayer.new()
	player.bus = bus
	add_child(player)
	return player


func _mute_bus(bus: StringName, mute: bool) -> void:
	AudioServer.set_bus_mute(AudioServer.get_bus_index(bus), mute)


func _stream(sound: String) -> AudioStream:
	if _streams.has(sound):
		return _streams[sound]
	var path: String = _AUDIO_DIR + sound + ".wav"
	var stream: AudioStream = load(path) as AudioStream if ResourceLoader.exists(path) else null
	if stream == null:
		if not _warned.has(sound):
			_warned[sound] = true
			push_warning("AudioManager: no sound at %s" % path)
		return null
	_streams[sound] = stream
	return stream


# Generated .wav files carry no loop points, so a loop is set to repeat when it is first played.
func _set_looping(stream: AudioStream) -> void:
	if stream is AudioStreamWAV:
		var wav: AudioStreamWAV = stream
		wav.loop_mode = AudioStreamWAV.LOOP_FORWARD
		wav.loop_begin = 0
		wav.loop_end = int(wav.get_length() * wav.mix_rate)
```
